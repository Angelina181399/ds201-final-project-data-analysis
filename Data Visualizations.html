import matplotlib.pyplot as plt

fig, ax1 = plt.subplots(figsize=(13, 5))

ax1.plot(df["period"], df["national_avg_fare"], color="tab:blue", linewidth=2)
ax1.set_ylabel("Average fare ($)", color="tab:blue")
ax1.tick_params(axis="y", labelcolor="tab:blue")

ax2 = ax1.twinx()
ax2.plot(df["period"], df["passengers_millions"], color="tab:orange", linewidth=2)
ax2.set_ylabel("Passengers (millions)", color="tab:orange")
ax2.tick_params(axis="y", labelcolor="tab:orange")

# --- COVID shading ---
start = df.index[df["period"] == "2020-Q1"][0]
end = df.index[df["period"] == "2021-Q2"][0]
ax1.axvspan(start, end, color="gray", alpha=0.2)
ax1.text((start + end) / 2, ax1.get_ylim()[1] * 0.97, "COVID-19",
         ha="center", va="top", fontsize=11, fontweight="bold", color="dimgray")

ax1.set_xticks(range(0, len(df), 4))
ax1.set_xticklabels(df["period"][::4], rotation=45)
plt.title("Average Domestic Airfare vs. Domestic Passengers by Quarter")
fig.tight_layout()
plt.show()

import numpy as np
import matplotlib.pyplot as plt

x = np.arange(4)   # Q1-Q4
w = 0.38           # bar width

fig, ax1 = plt.subplots(figsize=(8, 5))

# Fare bars (left axis)
b1 = ax1.bar(x - w/2, seasonal["median_fare"], w, color="tab:blue", label="Median fare ($)")
ax1.set_ylabel("Median fare ($)", color="tab:blue")
ax1.tick_params(axis="y", labelcolor="tab:blue")
ax1.set_ylim(0, seasonal["median_fare"].max() * 1.2)

# Passenger bars (right axis)
ax2 = ax1.twinx()
b2 = ax2.bar(x + w/2, seasonal["median_passengers"], w, color="tab:orange", label="Median passengers (millions)")
ax2.set_ylabel("Median passengers (millions)", color="tab:orange")
ax2.tick_params(axis="y", labelcolor="tab:orange")
ax2.set_ylim(0, seasonal["median_passengers"].max() * 1.2)

ax1.set_xticks(x)
ax1.set_xticklabels(["Q1", "Q2", "Q3", "Q4"])
ax1.legend([b1, b2], ["Median fare ($)", "Median passengers (millions)"], loc="upper left")

plt.title("Seasonal Pattern: Median Fare and Passengers by Quarter")
fig.tight_layout()
plt.show()
